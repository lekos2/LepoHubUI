local RemoteBlockActive = true

local function installRemoteBlock()
    if not (getrawmetatable and setreadonly and getnamecallmethod and newcclosure) then
        warn("[LEPO] Remote Block: executor sem suporte")
        return nil
    end
    local mt = getrawmetatable(game)
    local oldNamecall = mt.__namecall
    setreadonly(mt, false)
    local newFn = newcclosure(function(self, ...)
        if not RemoteBlockActive then
            return oldNamecall(self, ...)
        end
        local method = getnamecallmethod()
        if method == "InvokeServer" or method == "FireServer" then
            local name = ""
            pcall(function() name = tostring(self.Name or "") end)
            if BlockedRemotes[name] then
                return nil
            end
        end
        return oldNamecall(self, ...)
    end)
    mt.__namecall = newFn
    setreadonly(mt, true)
    return newFn
end

local remoteBlockHook = installRemoteBlock()

task.spawn(function()
    while RemoteBlockActive do
        task.wait(2)
        pcall(function()
            if not remoteBlockHook then
                remoteBlockHook = installRemoteBlock()
            end
        end)
    end
end)

local function afGetRF(name)
    local ok, rf = pcall(function()
        return AF_RS.Packages._Index["sleitnick_knit@1.7.0"].knit.Services.BallService.RF[name]
    end)
    if ok and rf then return rf end
    return nil
end

local AF_CreateHitbox = afGetRF("CreateHitbox")
local AF_Interact = afGetRF("Interact")
local AF_UpdateTilt = afGetRF("UpdateTilt")

local function afIsVisible(gui)
    if not gui then return false end
    local pg = AF_Player:FindFirstChildOfClass("PlayerGui")
    while gui and gui ~= pg do
        if gui:IsA("GuiObject") and not gui.Visible then return false end
        if gui:IsA("ScreenGui") and not gui.Enabled then return false end
        gui = gui.Parent
    end
    return true
end

local function afFindUI(path)
    local pg = AF_Player:FindFirstChildOfClass("PlayerGui")
    if not pg then return nil end
    for seg in tostring(path):gmatch("[^%.]+") do
        pg = pg and pg:FindFirstChild(seg)
    end
    if pg and (pg:IsA("TextButton") or pg:IsA("ImageButton")) then
        return pg
    end
    return nil
end

local function afFireButton(btn)
    if not btn then return false end
    if type(getconnections) ~= "function" then return false end
    local ok, cons = pcall(getconnections, btn.Activated)
    if not ok or not cons then return false end
    for _, c in ipairs(cons) do
        if c.Fire then
            if pcall(function() c:Fire() end) then return true end
        end
    end
    return false
end

local function afFindPlayPad()
    local lobby = AF_WS:FindFirstChild("Volleyball Lobby")
    lobby = lobby and lobby:FindFirstChild("Interactables")
    lobby = lobby and lobby:FindFirstChild("PLAY")
    if not lobby then return nil end
    return lobby:FindFirstChild("Ring")
        or lobby:FindFirstChild("Glow")
        or lobby:FindFirstChildWhichIsA("BasePart", true)
end

local function afWalkTo(targetPos)
    local char = AF_Player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    hum:MoveTo(targetPos)
end

local function afEmPartida()
    return AF_Player:GetAttribute("Gameplay_InGame") == true
end

local function afGetBall()
    for _, v in ipairs(AF_WS:GetDescendants()) do
        if typeof(v.Name) == "string" and v.Name:sub(1, 12) == "CLIENT_BALL_" then
            local part = v:IsA("BasePart") and v or v:FindFirstChildWhichIsA("BasePart", true)
            if part then
                local id = v:GetAttribute("ServerId")
                    or part:GetAttribute("ServerId")
                    or v:GetAttribute("Id")
                    or part:GetAttribute("Id")
                    or tonumber(v.Name:sub(13))
                if id then return part, id end
            end
        end
    end
    return nil, 0
end

local function afGetHRP()
    local c = AF_Player.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function afGetTimestamp()
    local ok, t = pcall(function() return AF_WS:GetServerTimeNow() end)
    if ok and typeof(t) == "number" then return t end
    return tick()
end

local function afTick()
    if not AF.enabled then return end
    if os.clock() - AF.lastCheck < 1.2 then return end
    AF.lastCheck = os.clock()

    local backBtn = afFindUI("Interface.RoundOverStats.ReturnContainer.BackBtn")
    if afIsVisible(backBtn) then
        afFireButton(backBtn)
        return
    end

    local pg = AF_Player:FindFirstChildOfClass("PlayerGui")
    local interface = pg and pg:FindFirstChild("Interface")
    local teamSel = interface and interface:FindFirstChild("TeamSelection")

    if afIsVisible(teamSel) then
        local slot = nil
        for _, t in ipairs({"1", "2"}) do
            local teamHolder = teamSel:FindFirstChild(t)
            teamHolder = teamHolder and teamHolder:FindFirstChild("TeamHolder")
            if teamHolder then
                for _, child in ipairs(teamHolder:GetChildren()) do
                    if child:IsA("GuiButton") and afIsVisible(child) then
                        local un = child:FindFirstChild("Username")
                        local hs = child:FindFirstChild("Headshot")
                        local empty = (not un or tostring(un.Text) == "")
                            and (not hs or hs.Visible == false)
                        if empty then slot = child break end
                    end
                end
            end
            if slot then break end
        end
        if slot then
            afFireButton(slot)
            return
        end
        return
    end

    if AF_Player:GetAttribute("Gameplay_InGame") == true then
        return
    end

    local pad = afFindPlayPad()
    local char = AF_Player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not pad or not root then return end

    local dist = (Vector3.new(pad.Position.X, root.Position.Y, pad.Position.Z) - root.Position)
        * Vector3.new(1, 0, 1)

    if dist.Magnitude > 4 then
        afWalkTo(pad.Position)
    else
        if (AF.lastTouch or 0) <= os.clock() then
            AF.lastTouch = os.clock() + 3
            if type(getconnections) == "function" then
                pcall(function()
                    for _, c in ipairs(getconnections(pad.Touched)) do
                        c:Fire(root)
                    end
                end)
            end
        end
    end
end

RunService.Heartbeat:Connect(function()
    if not AF.enabled then return end
    if not afEmPartida() then return end
    local ball = select(1, afGetBall())
    local hrp = afGetHRP()
    if ball and hrp then
        pcall(function() hrp.CFrame = ball.CFrame end)
    end
end)

task.spawn(function()
    while true do
        task.wait(AF.interval or 0.01)
        if AF.enabled then
            pcall(afTick)
            if afEmPartida() then
                local ball, bid = afGetBall()
                local hrp = afGetHRP()
                if ball and bid ~= 0 and AF_CreateHitbox and AF_Interact then
                    local look = AF_Cam and AF_Cam.CFrame.LookVector or Vector3.new(0, 0, -1)
                    local move = hrp and hrp.CFrame.LookVector or Vector3.zero
                    local tilt = Vector3.new(0, 1, 0)
                    local ts = afGetTimestamp()
                    local key = "Block" .. math.random(1000000, 9999999)

                    pcall(function()
                        AF_CreateHitbox:InvokeServer({
                            BallId = bid, Charge = 1, Key = key,
                            ClientTimestamp = ts, Move = "Block"
                        })
                    end)

                    pcall(function()
                        AF_Interact:InvokeServer({
                            Charge = 1, Move = "Block", SpecialCharge = 0,
                            TiltDirection = tilt, BallId = bid, MoveDirection = move,
                            Key = key, From = "Client", LookVector = look,
                            Timestamp = ts, ClientCanRunSpecial = false, CustomClient = {}
                        })
                    end)

                    if AF_UpdateTilt then
                        pcall(function() AF_UpdateTilt:InvokeServer(tilt) end)
                    end
                end
            end
        end
    end
end)

local function readLuckyCount()
    local now = os.clock()
    if now - (G.luckyCache and G.luckyCache.stamp or 0) < 0.3 and G.luckyCache and G.luckyCache.value ~= nil then
        return G.luckyCache.value
    end
    G.luckyCache = G.luckyCache or { value = nil, stamp = 0 }
    for _, name in ipairs({ "LuckySpins", "LuckySpin", "Lucky", "Luckys" }) do
        local n = tonumber(Player:GetAttribute(name))
        if n then G.luckyCache = { value = n, stamp = now } return n end
    end
    local ls = Player:FindFirstChild("leaderstats")
    if ls then
        for _, name in ipairs({ "LuckySpins", "LuckySpin", "Lucky", "Luckys" }) do
            local obj = ls:FindFirstChild(name)
            if obj and tonumber(obj.Value) then
                local n = tonumber(obj.Value)
                G.luckyCache = { value = n, stamp = now }
                return n
            end
        end
    end
    local pg = Player:FindFirstChild("PlayerGui")
    if not pg then return nil end
    local hasBuyButton = false
    for _, obj in ipairs(pg:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") then
            local txt = obj.Text or ""
            local upper = string.upper(txt)
            local num = string.match(txt, "[Ll][Uu][Cc][Kk][Yy]%s+[Ss][Pp][Ii][Nn][Ss]%s*%-%s*(%d+)")
            if num then
                local n = tonumber(num)
                G.luckyCache = { value = n, stamp = now }
                return n
            end
            if string.find(upper, "BUY LUCKY SPINS", 1, true) then hasBuyButton = true end
        end
    end
    if hasBuyButton then G.luckyCache = { value = 0, stamp = now } return 0 end
    return nil
end

local function doSpin(useLucky)
    local StyleService = ReplicatedStorage.Packages._Index["sleitnick_knit@1.7.0"].knit.Services.StyleService
    if not StyleService then return false end
    local RF = StyleService:FindFirstChild("RF")
    if not RF then return false end
    local Roll = RF:FindFirstChild("Roll")
    if not Roll then return false end
    local ok = pcall(function() Roll:InvokeServer(useLucky and true or false) end)
    if not ok then pcall(function() Roll:InvokeServer({ Lucky = useLucky }) end) end
    return true
end

local function doAbilitySpin(useLucky)

    local ok = pcall(function() Roll:InvokeServer(useLucky and true or false) end)
    if not ok then pcall(function() Roll:InvokeServer({ Lucky = useLucky }) end) end
    return true
end

local function matchesTarget(current, target)
    if not current or not target or target == "" then return false end
    return string.lower(tostring(current)) == string.lower(tostring(target))
end

local function startAutoSpin()
    task.spawn(function()
        while G.autoSpin do
            local cur = readCurrentStyle()
            if cur == G.targetStyle then G.autoSpin = false break end
            doSpin(G.styleSpinType == "Lucky")
            task.wait(0.1)
        end
    end)
end

local function startAutoAbilitySpin()
    task.spawn(function()
        while G.autoAbilitySpin do
            local cur = readCurrentAbility()
            if cur == G.targetAbility then G.autoAbilitySpin = false break end
            doAbilitySpin(G.abilitySpinType == "Lucky")
            task.wait(0.1)
        end
    end)
end

local function startAutoRollback()
    if G.AutoRollback and G.AutoRollback.running then return end
    G.AutoRollback = G.AutoRollback or { enabled = false, targetStyle = nil, targetAbility = nil, running = false }
    G.AutoRollback.running = true
    G.luckyCache = { value = nil, stamp = 0 }
    G.luckyAbilityCache = { value = nil, stamp = 0 }

    local function finishStop()
        G.AutoRollback.enabled = false
        G.AutoRollback.running = false
    end

    local function triggerRB()
        G.AutoRollback.running = false
        G.AutoRollback.enabled = false
        task.wait(0.4)
        triggerDataRollback()
    end

    task.spawn(function()
        local targetStyle = G.AutoRollback.targetStyle
        local targetAbility = G.AutoRollback.targetAbility
        local wantStyle = targetStyle and targetStyle ~= "" and targetStyle ~= "(Nenhum)"
        local wantAbility = targetAbility and targetAbility ~= "" and targetAbility ~= "(Nenhuma)"
        if not wantStyle and not wantAbility then finishStop() return end

        if wantStyle then
            local cur = readCurrentStyle()
            if not matchesTarget(cur, targetStyle) then
                local safety = 0
                while G.AutoRollback.enabled and safety < 500 do
                    safety = safety + 1
                    cur = readCurrentStyle()
                    if matchesTarget(cur, targetStyle) then break end
                    local luckies = readLuckyCount()
                    if luckies ~= nil and luckies <= 0 then triggerRB() return end
                    doSpin(true)
                    task.wait(0.5)
                end
            end
            if not G.AutoRollback.enabled then finishStop() return end
            if not wantAbility then finishStop() return end
        end

        if wantAbility and G.AutoRollback.enabled then
            local cur2 = readCurrentAbility()
            if not matchesTarget(cur2, targetAbility) then
                local safety = 0
                while G.AutoRollback.enabled and safety < 500 do
                    safety = safety + 1
                    cur2 = readCurrentAbility()
                    if matchesTarget(cur2, targetAbility) then break end
                    local luckies = readLuckyAbilityCount()
                    if luckies ~= nil and luckies <= 0 then triggerRB() return end
                    doAbilitySpin(true)
                    task.wait(0.5)
                end
            end
        end
        finishStop()
    end)
end

local function gpGetNearestBall()
    local nearest, dist = nil, math.huge
    local c = _GP_LP.Character
    if not c then return nil, nil end
    local r = c:FindFirstChild("HumanoidRootPart")
    if not r then return nil, nil end
    for _, m in ipairs(_GP_WS:GetChildren()) do
        if m:IsA("Model") and m.Name:match("^CLIENT_BALL_%d+$") then
            for _, p in ipairs(m:GetDescendants()) do
                if p:IsA("BasePart") then
                    local d = (p.Position - r.Position).Magnitude
                    if d < dist then dist = d nearest = p end
                    break
                end
            end
        end
    end
    return nearest, dist
end

local AC = G.autoCorner
AC.physics = function()
    local ok, mod = pcall(require, _GP_RS.Common.Physics)
    return ok and mod or nil
end
AC.bounds = function()
    local phy = AC.physics()
    if phy and phy._getCourtlines then
        local ok, lines = pcall(phy._getCourtlines)
        if ok and lines and lines:IsA("BasePart") then
            return lines.Position, lines.Size * 0.5, lines.Position.Y
        end
    end
    local map = _GP_WS:FindFirstChild("Map")
    local court = map and map:FindFirstChild("Court")
    if court then
        return court.Position, court.Size * 0.5 - Vector3.new(1, 0, 1), court.Position.Y + court.Size.Y * 0.5
    end
    return nil
end
AC.trackBall = function(part, prevPos, dt)
    if dt <= 0 then return end
    AC.drift = (part.Position - prevPos) / dt
end
AC.origin = function()
    local ball = gpGetNearestBall()
    local c = _GP_LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    if not ball then return root and root.Position or Vector3.zero end
    local drift = AC.drift
    if not root or typeof(drift) ~= "Vector3" or drift.Magnitude < 1 then return ball.Position end
    local gap = ball.Position - root.Position
    local closing = drift - root.AssemblyLinearVelocity
    local speed = closing:Dot(closing)
    if speed < 1 then return ball.Position end
    local reach = math.clamp(-gap:Dot(closing) / speed, 0, 0.45)
    return ball.Position + drift * reach
end

local function gpCornerPoint()
    local c = _GP_LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    local centre, half, surfaceY = AC.bounds()
    if not root or not centre then return nil end
    local halfX = math.max(half.X - 2, 1)
    local halfZ = math.max(half.Z - 2, 1)
    local map = _GP_WS:FindFirstChild("Map")
    local collide = map and map:FindFirstChild("BallCollideOnly")
    local net = collide and collide:FindFirstChild("Net")
    local netZ = net and net.Position.Z or centre.Z
    local idx = _GP_LP.Team and _GP_LP.Team:GetAttribute("Index")
    local sign
    if idx == 1 then sign = 1 elseif idx == 2 then sign = -1
    else sign = (root.Position.Z < netZ) and 1 or -1 end
    local deepZ = centre.Z + sign * halfZ
    local leftX = centre.X + sign * halfX
    local rightX = centre.X - sign * halfX
    local pivot = AC.origin()
    local standoff = math.abs(root.Position.Z - netZ)
    local hugging = 1 - math.clamp(standoff / (half.Z / 3), 0, 1)
    AC.lean = hugging
    local shortZ = centre.Z + sign * halfZ * 0.3
    local reach = AC.reach
    local wantsNear = reach == "Nearest" or (reach ~= "Back" and hugging > 0)
    local side = AC.side
    if side == "Left" then return Vector3.new(leftX, surfaceY, wantsNear and shortZ or deepZ)
    elseif side == "Right" then return Vector3.new(rightX, surfaceY, wantsNear and shortZ or deepZ) end
    local corners = {}
    if reach ~= "Nearest" then
        corners[#corners + 1] = { at = Vector3.new(leftX, surfaceY, deepZ), reward = 0 }
        corners[#corners + 1] = { at = Vector3.new(rightX, surfaceY, deepZ), reward = 0 }
    end
    if wantsNear then
        corners[#corners + 1] = { at = Vector3.new(leftX, surfaceY, shortZ), reward = 0.45 * hugging }
        corners[#corners + 1] = { at = Vector3.new(rightX, surfaceY, shortZ), reward = 0.45 * hugging }
    end
    if #corners == 0 then return Vector3.new(leftX, surfaceY, deepZ) end
    local heading = _GP_CAM and _GP_CAM.CFrame.LookVector or root.CFrame.LookVector
    heading = Vector3.new(heading.X, 0, heading.Z)
    if heading.Magnitude < 0.01 then return corners[1].at end
    heading = heading.Unit
    local aligned = {}
    for _, cc in ipairs(corners) do
        local line = Vector3.new(cc.at.X - pivot.X, 0, cc.at.Z - pivot.Z)
        if line.Magnitude > 0.01 then
            aligned[#aligned + 1] = { spot = cc.at, dot = line.Unit:Dot(heading), reward = cc.reward }
        end
    end
    if #aligned == 0 then return corners[1].at end
    local pick
    for _, e in ipairs(aligned) do if not pick or e.dot > pick.dot then pick = e end end
    local wanted = pick.spot.X >= centre.X
    local best
    for _, e in ipairs(aligned) do
        if (e.spot.X >= centre.X) == wanted then
            if not best or e.reward > (best.score or 0) then best = { spot = e.spot, score = e.reward } end
        end
    end
    return best and best.spot or pick.spot
end

AC.facing = function()
    local spot = gpCornerPoint()
    if not spot then return nil end
    local pivot = AC.origin()
    local flat = Vector3.new(spot.X - pivot.X, 0, spot.Z - pivot.Z)
    local root = _GP_LP.Character and _GP_LP.Character:FindFirstChild("HumanoidRootPart")
    if root then
        local map = _GP_WS:FindFirstChild("Map")
        local collide = map and map:FindFirstChild("BallCollideOnly")
        local net = collide and collide:FindFirstChild("Net")
        local netZ = net and net.Position.Z
        if netZ then
            local sign = (netZ < root.Position.Z) and 1 or -1
            if flat.Z * sign >= 0 then flat = Vector3.new(spot.X - root.Position.X, 0, spot.Z - root.Position.Z) end
            if flat.Z * sign >= 0 then flat = Vector3.new(flat.X, 0, -sign * math.max(math.abs(flat.Z), 1)) end
        end
    end
    if flat.Magnitude < 0.01 then return nil end
    return flat.Unit
end

AC.installAimed = function()
    if AC.aimed then return end
    if not (getrawmetatable and setreadonly and getnamecallmethod) then return end
    local mt = getrawmetatable(game)
    local old = mt.__namecall
    setreadonly(mt, false)
    mt.__namecall = function(self, ...)
        local m = getnamecallmethod()
        if self == _GP_BALL.RF.Interact and m == "InvokeServer" then
            local d = (...)
            if AC.enabled and type(d) == "table" and (d.Move == "Spike" or d.Move == "ServeSpike") then
                local steer = AC.steer
                if typeof(steer) == "Vector3" then
                    local flat = Vector3.new(steer.X, 0, steer.Z)
                    if flat.Magnitude > 0.01 then
                        flat = flat.Unit
                        local aim = (flat - Vector3.new(0, 0.35, 0)).Unit
                        d.LookVector = aim
                        d.Direction = aim
                    end
                end
            end
            return old(self, d)
        end
        return old(self, ...)
    end
    setreadonly(mt, true)
    AC.aimed = true
end

_GP_UIS.JumpRequest:Connect(function()
    if not AC.enabled then return end

    if not c then return end

    if not root then return end
    local ball = gpGetNearestBall()

    local t0 = os.clock()
    task.defer(function()
        task.wait(0.03)

        if steer and steer.Magnitude > 0 then
            AC.steer = steer
            root.CFrame = CFrame.lookAt(root.Position, root.Position + steer.Unit)
        end
    end)
end)

    return (root.Position.Z >= netZ) and 1 or -1
end
local function ctrFindThreat(root, net)
    local netZ = net.Position.Z
    local netTop = net.Position.Y + net.Size.Y * 0.5
    local side = ctrOwnSide(root, netZ)
    local myGap = (root.Position.Z - netZ) * side
    if myGap < 0 or myGap > CTR.netRange then return nil end
    local ball = gpGetNearestBall()
    if not ball then return nil end
    if ball.Position.Y < netTop - 3 then return nil end
    if math.abs(ball.Position.Z - netZ) > CTR.netRange + 6 then return nil end
    if math.abs(ball.Position.X - root.Position.X) > CTR.lateral then return nil end
    for _, p in ipairs(_GP_PLR:GetPlayers()) do
        local hostile = p ~= _GP_LP and (_GP_LP.Team == nil or p.Team ~= _GP_LP.Team)
        if hostile then
            local cc = p.Character
            local eR = cc and cc:FindFirstChild("HumanoidRootPart")
            if eR and cc:GetAttribute("Jumping") then
                local eG = (eR.Position.Z - netZ) * side
                if eG < 0 and -eG <= CTR.netRange + 4
                    and (ball.Position - eR.Position).Magnitude <= CTR.reach
                    and math.abs(eR.Position.X - root.Position.X) <= CTR.lateral then
                    return ball, netZ, side
                end
            end
        end
    end
    return nil
end
local function ctrPickMove(ball, netZ, side)
    if CTR.move ~= "Auto" then return CTR.move end
    if (ball.Position.Z - netZ) * side > 0 then return "Spike" end
    return "Block"
end
local function ctrFire(move, ball)
    if CTR.firing then return end
    local c = _GP_LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    if not root then return end
    CTR.firing = true
    task.spawn(function()
        if CTR.autoJump then
            pcall(function()
                local hum = c:FindFirstChildOfClass("Humanoid")
                if hum then hum.Jump = true end
            end)
        end
        if CTR.delay > 0 then task.wait(CTR.delay) end
        if move == "Spike" and ball then
            local ok = false
            if _GP_BALL and _GP_BALL.RF and _GP_BALL.RF.Interact then
                pcall(function()
                    local d = {
                        Action = "Spike", Move = "Spike",
                        Charge = 1, SpecialCharge = 1e-06,
                        LookVector = root.CFrame.LookVector,
                        TiltDirection = root.CFrame.LookVector,
                        MoveDirection = root.CFrame.LookVector,
                        From = "Client",
                    }
                    if _GP_BALL.RF.Interact:IsA("RemoteFunction") then
                        _GP_BALL.RF.Interact:InvokeServer(d)
                    else
                        _GP_BALL.RF.Interact:FireServer(d)
                    end
                    ok = true
                end)
            end
            if not ok then
                local vim = VirtualInputManager
                if vim then
                    vim:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
                    task.wait(0.03)
                    vim:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
                end
            end
        else
            local vim = VirtualInputManager
            if vim then
                vim:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
                task.wait(0.03)
                vim:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
            end
        end
        CTR.lastFire = os.clock()
        CTR.firing = false
    end)
end
_GP_RUN.Heartbeat:Connect(function()
    if not CTR.enabled or CTR.firing then return end
    if os.clock() - CTR.lastFire < CTR.cooldown then return end
    local c = _GP_LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local net = ctrGetNet()
    if not net then return end
    local ball, netZ, side = ctrFindThreat(root, net)
    if not ball then return end
    ctrFire(ctrPickMove(ball, netZ, side), ball)
end)

local function AS_isSameTeam(p)
    local team = _GP_LP.Team
    if not team then return false end
    return p.Team == team
end
local function AS_getTarget(mode)
    local char = _GP_LP.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local candidates = {}
    for _, p in ipairs(_GP_PLR:GetPlayers()) do
        if p ~= _GP_LP and AS_isSameTeam(p) then
            local c = p.Character
            if c then
                local r = c:FindFirstChild("HumanoidRootPart")
                local h = c:FindFirstChildOfClass("Humanoid")
                if r and h and h.Health > 0 then table.insert(candidates, r) end
            end
        end
    end
    if #candidates == 0 then return nil end
    if mode == "Random" then return candidates[math.random(1, #candidates)] end
    local best, bestDist = nil, math.huge
    for _, r in ipairs(candidates) do
        local d = (r.Position - root.Position).Magnitude
        if d < bestDist then best = r bestDist = d end
    end
    return best
end
local function AS_getBallId()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if typeof(obj.Name) == "string" and obj.Name:sub(1, 12) == "CLIENT_BALL_" then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
            if part then
                local id = obj:GetAttribute("ServerId") or part:GetAttribute("ServerId")
                    or obj:GetAttribute("Id") or part:GetAttribute("Id") or tonumber(obj.Name:sub(13))
                if id then return id end
            end
        end
    end
    return nil
end
local function AS_calcAutoSet()
    local char = _GP_LP.Character
    if not char then G.autoSet.lookVector, G.autoSet.direction, G.autoSet.tilt = nil, nil, nil return end

    local target = AS_getTarget(G.autoSet.mode)
    if not target then G.autoSet.lookVector, G.autoSet.direction, G.autoSet.tilt = nil, nil, nil return end
    local flat = Vector3.new(target.Position.X - root.Position.X, 0, target.Position.Z - root.Position.Z)
    if flat.Magnitude < 0.01 then G.autoSet.lookVector, G.autoSet.direction, G.autoSet.tilt = nil, nil, nil return end

    G.autoSet.lookVector = Vector3.new(unit.X, 0, unit.Z)
    G.autoSet.direction = Vector3.new(unit.X, 0, unit.Z)
    G.autoSet.tilt = Vector3.new(unit.X, 1, unit.Z)
end

Kisuki.effect = function(isStrong)

Kisuki.install = function()
    if Kisuki.hooked then return end
    local ok, Knit = pcall(require, _GP_RS.Packages.Knit)
    if not ok then return end
    local found, ctrl = pcall(Knit.GetController, "GameController")
    if not found or type(ctrl) ~= "table" or type(ctrl.Dive) ~= "function" then return end
    local originalDive = ctrl.Dive
    ctrl.Dive = function(self, params, ...)
        if not Kisuki.enabled then return originalDive(self, params, ...) end
        local chargeBlend = math.clamp(Kisuki.charge, 0, 1)
        local styled, special = pcall(require, _GP_RS.Content.Style.Kimiro.Special)
        local tuning = styled and type(special) == "table" and special.Metadata
        if type(tuning) == "table" then
            params = type(params) == "table" and params or {}
            params.MaxForce = tuning.MaxForce
            params.TimeScale = tuning.TimeScale
            params.StretchSpeedFactor = tuning.StretchSpeedFactor
            params.EasingStyle = tuning.EasingStyle
            params.Debounce = tuning.Debounce
            if not params.Target and typeof(tuning.Power) == "NumberRange" then
                local char = _GP_LP.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                local moveDir = hum and hum.MoveDirection or Vector3.zero
                if moveDir.Magnitude < 0.5 and char then moveDir = char:GetPivot().LookVector end
                local flat = Vector3.new(moveDir.X, 0, moveDir.Z)
                if flat.Magnitude > 0.01 then
                    params.Target = flat.Unit * (tuning.Power.Min + (tuning.Power.Max - tuning.Power.Min) * chargeBlend)
                end
            end
        end
        local result = originalDive(self, params, ...)
        if result ~= nil then Kisuki.effect(chargeBlend >= 0.5) end
        return result
    end
    Kisuki.hooked = true
end

    local gameController = nil
    pcall(function()

    end)

            or humanoid.FloorMaterial ~= Enum.Material.Air or os.clock() >= expiresAt then

            if LeadFeetConnection then LeadFeetConnection:Disconnect() LeadFeetConnection = nil end
            return
        end
        local velocity = root.AssemblyLinearVelocity
        local nextY = math.max(velocity.Y + (metadata.Acceleration or -10000) * delta, metadata.TerminalVelocity or -300)
        root.AssemblyLinearVelocity = Vector3.new(velocity.X, nextY, velocity.Z)
    end)
end

        if not character or not character.PrimaryPart then return end
        local root = character.PrimaryPart
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 0 then return end
        local metadata = { Acceleration = -10000, TerminalVelocity = -300, Duration = 2 }
        local gameController = nil
        pcall(function()
            local abilityModule = require(ReplicatedStorage.Content.Ability.LeadFeet)
            if type(abilityModule) == "table" and type(abilityModule.Metadata) == "table" then metadata = abilityModule.Metadata end
            gameController = require(ReplicatedStorage.Packages.Knit).GetController("GameController")
            if gameController then gameController.IsBusy:set(false) end
            require(ReplicatedStorage.Content.Effect):Play("LeadFeetFX")({ Player = Player })
            require(ReplicatedFirst.Controllers.GameController.Actions.Move.Jump).stop()
            if gameController then gameController.IsBusy:set(true) end
        end)
        if LeadFeetConnection then LeadFeetConnection:Disconnect() LeadFeetConnection = nil end
        local expiresAt = os.clock() + (metadata.Duration or 2)
        LeadFeetConnection = RunService.Heartbeat:Connect(function(delta)
            if not root.Parent or not humanoid.Parent or humanoid.Health <= 0
                or humanoid.FloorMaterial ~= Enum.Material.Air or os.clock() >= expiresAt then
                if gameController then pcall(function() gameController.IsBusy:set(false) end) end
                if LeadFeetConnection then LeadFeetConnection:Disconnect() LeadFeetConnection = nil end
                return
            end
            local velocity = root.AssemblyLinearVelocity
            local nextY = math.max(velocity.Y + (metadata.Acceleration or -10000) * delta, metadata.TerminalVelocity or -300)
            root.AssemblyLinearVelocity = Vector3.new(velocity.X, nextY, velocity.Z)
        end)
    end)
    return LeedFeatTouchButton
end

local function gpLoadBallSkins()
    G.ballList = {}
    local A = _GP_RS:FindFirstChild("Assets")
    local B = A and A:FindFirstChild("Ball")
    if B then
        for _, c in ipairs(B:GetChildren()) do
            if c:IsA("Model") then table.insert(G.ballList, c.Name) end
        end
    end
    table.sort(G.ballList)
    if #G.ballList == 0 then table.insert(G.ballList, "ClassicBall") end
    return G.ballList
end
gpLoadBallSkins()

local function gpSweepBallSkin()
    local bf = _GP_RS:FindFirstChild("Assets")
    bf = bf and bf:FindFirstChild("Ball")
    if not bf then return end
    local function strip(m)
        local h = G.clones[m]
        if h then h:Destroy() G.clones[m] = nil end
        local s = G.originals[m]
        if s then
            for p, r in pairs(s) do
                if p.Parent then pcall(function() p[r.field] = r.value end) end
            end
            G.originals[m] = nil
        end
    end
    local function skin(m)
        local d = m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart")
        local src = bf:FindFirstChild(G.ballSkin)
        if not d or not src then return end
        strip(m)
        local saved = {}
        local function st(p, f, v) saved[p] = {field=f, value=p[f]} p[f] = v end
        for _, p in ipairs(m:GetDescendants()) do
            if p:IsA("BasePart") then st(p, "Transparency", 1) end
        end
        G.originals[m] = saved
        local h = src:Clone()
        h.Parent = m
        local pieces = {}
        if h:IsA("BasePart") then pieces[#pieces+1] = h end
        for _, c in ipairs(h:GetDescendants()) do
            if c:IsA("BasePart") then pieces[#pieces+1] = c end
        end
        h:PivotTo(d.CFrame)
        for _, p in ipairs(pieces) do
            p.Anchored = false p.CanCollide = false p.Massless = true
            local w = Instance.new("WeldConstraint")
            w.Part0 = d w.Part1 = p w.Parent = p
        end
        G.clones[m] = h
    end
    for m in pairs(G.clones) do strip(m) end
    if not G.ballChanger then return end
    for _, m in ipairs(_GP_WS:GetChildren()) do
        if m:IsA("Model") and m.Name:match("^CLIENT_BALL_") then skin(m) end
    end
    if not G.seen.__ballHook then
        G.seen.__ballHook = _GP_WS.ChildAdded:Connect(function(child)
            if not G.ballChanger or not child:IsA("Model") or not child.Name:match("^CLIENT_BALL_") then return end
            task.spawn(function()
                for _ = 1, 40 do
                    if child.PrimaryPart or child:FindFirstChildWhichIsA("BasePart") then break end
                    task.wait(0.05)
                end
                if G.ballChanger and child.Parent then gpSweepBallSkin() end
            end)
        end)
    end
end

local function gpFireCardRemote()
    if G.isProcessingCard then return end
    local Remote = _GP_RS.Packages._Index["sleitnick_knit@1.7.0"].knit.Services.PlayerCardService.RE.OnFlashCard
    if Remote and firesignal then
        G.isProcessingCard = true
        pcall(firesignal, Remote.OnClientEvent, {
            ["PlayerCardItemId"] = G.playerCardSel,
            ["Context"] = { ["Custom"] = {}, ["PlacementId"] = "ScoredPoint" },
            ["Player"] = _GP_LP,
        })
        task.wait(0.3)
        G.isProcessingCard = false
    end
end

local function gpChangePlayerCard()
    if not G.playerCard or G.isProcessingCard then return end
    local flashFrame = _GP_LP.PlayerGui.Interface.Persistent:FindFirstChild("FlashPlayerCardFrame")
    if not flashFrame then return end
    local username = flashFrame:FindFirstChild("Username", true)
    if username and username:IsA("TextLabel") then
        local currentText = string.upper(username.Text or "")
        local playerName = string.upper(_GP_LP.Name)
        if string.find(currentText, playerName) then
            flashFrame:Destroy()
            task.wait(0.1)
            gpFireCardRemote()
        end
    end
end

local function gpEnableHitEffect()
    if G.hitEffectConn1 then return end
    G.hitEffectFired = false
    G.hitEffectLocal = false
    local Remote = _GP_RS.Packages._Index["sleitnick_knit@1.7.0"].knit.Services.BallService.RE.HitGround
    
    G.hitEffectConn3 = Remote.OnClientEvent:Connect(function(...)
        local args = {...}
        local myName = _GP_LP.Name
        for _, val in pairs(args) do
            if tostring(val) == myName then 
                G.hitEffectLocal = true 
                break 
            end
        end
    end)

    local function fireHitRemote(hitPos)
        if G.hitEffectFired or not G.hitEffectLocal then return end
        if not G.scoreEffect then return end
        G.hitEffectFired = true
        
        local myUsername = _GP_LP.Name
        local targetObject = workspace:FindFirstChild(myUsername)
        local effectId = G.scoreEffectSel
        local hardHit = true
        
        if effectId == "GroundHit1" or effectId == "GroundHit2" then
            hardHit = effectId == "GroundHit2"
            effectId = nil
        end

        if firesignal then
            firesignal(Remote.OnClientEvent, table.unpack({
                hitPos, hardHit, false, 5, effectId, targetObject,
                Vector3.new(-0.000002088591600113432, 1, -0.000004710930170404026)
            }, 1, 7))
        end
        
        task.wait(0.5)
        G.hitEffectFired = false
        G.hitEffectLocal = false
    end

    G.hitEffectConn1 = workspace.ChildAdded:Connect(function(child)
        if child.Name == "HitIndicator" and child.Color == Color3.fromRGB(0, 255, 0) then 
            fireHitRemote(child.Position) 
        end
    end)
    
    G.hitEffectConn2 = workspace.DescendantAdded:Connect(function(child)
        if child.Name == "HitIndicator" and child:IsA("BasePart") and child.Color == Color3.fromRGB(0, 255, 0) then
            fireHitRemote(child.Position)
        end
    end)
end

local function gpDisableHitEffect()

                    line.Shape = Enum.PartType.Cylinder

                end

                local line = espCylinders[p]
                local len = LHUI.ESPLineSize or 10
                line.Material = LHUI.ESPNeon and Enum.Material.Neon or Enum.Material.SmoothPlastic

                local eyePos = hrp.Position + Vector3.new(0, 1.6, 0) + (flatLook * 0.65)
                local endPos = eyePos + flatLook * len
                local midPos = eyePos + flatLook * (len / 2)

                line.CFrame = CFrame.lookAt(midPos, endPos) * CFrame.Angles(0, math.rad(90), 0)

        if not p.Parent or not p.Character then

            espCylinders[p] = nil
        end
    end

local GP_GRAVITY = Vector3.new(0, -(Workspace.Gravity or 17), 0)
local GP_BALL_RADIUS = 1.2835
local GP_VEL_SMOOTH = 0.35
local GP_MIN_SPEED = 0.3
local GP_CACHE_TIMEOUT = 3.0
local GP_FLOOR_CACHE = nil

local gpTrailState = {
    trailParts = {},
    markerPart = nil,
    markerBillboard = nil,
    markerLabel = nil,
    lastBall = nil,
    lastPos = nil,
    lastTime = 0,
    velocity = Vector3.zero,
    velSamples = {},
    horizon = 4,
    cachedFall = nil,
    cachedFallT = nil,
    cachedFallAt = 0,
    markerTargetPos = nil,
}

local function gpGetBallForTrail()
    local nearest, dist = nil, math.huge
    local c = _GP_LP.Character
    if not c then return nil end
    local r = c:FindFirstChild("HumanoidRootPart")
    if not r then return nil end
    for _, m in ipairs(_GP_WS:GetChildren()) do
        if m:IsA("Model") and m.Name:match("^CLIENT_BALL_%d+$") then
            for _, p in ipairs(m:GetDescendants()) do
                if p:IsA("BasePart") then
                    local d = (p.Position - r.Position).Magnitude
                    if d < dist then dist = d nearest = p end
                    break
                end
            end
        end
    end
    return nearest
end

local function gpPhysModule()
    local ok, mod = pcall(require, _GP_RS.Common.Physics)
    if ok and type(mod) == "table" then return mod end
    return nil
end

local function gpFloorY(pos)
    local mod = gpPhysModule()
    if mod and mod.calculateFloorHeight then
        local ok, y = pcall(mod.calculateFloorHeight, pos, false)
        if ok and type(y) == "number" and y > -1000 then return y end
    end
    if GP_FLOOR_CACHE then return GP_FLOOR_CACHE end
    local map = _GP_WS:FindFirstChild("Map")
    if map then
        local floor = map:FindFirstChild("Floor")
            or map:FindFirstChild("Court")
            or map:FindFirstChild("Baseplate")
            or (map:FindFirstChild("BallNoCollide") and map.BallNoCollide:FindFirstChild("Floor"))
        if floor and floor:IsA("BasePart") then
            GP_FLOOR_CACHE = floor.Position.Y + floor.Size.Y * 0.5
            return GP_FLOOR_CACHE
        end
    end
    local c = _GP_LP.Character
    local h = c and c:FindFirstChild("HumanoidRootPart")
    if h then
        GP_FLOOR_CACHE = h.Position.Y - 3
        return GP_FLOOR_CACHE
    end
    return 0
end

local function gpProjectile(pos, vel, dt)
    local mod = gpPhysModule()
    if mod and mod.calculateProjectilePosition then
        local ok, p = pcall(mod.calculateProjectilePosition, pos, vel, GP_GRAVITY, dt, Vector3.zero)
        if ok and typeof(p) == "Vector3" then return p end
    end
    return pos + vel * dt + 0.5 * GP_GRAVITY * dt * dt
end

local function gpUpdateBallVel()
    local ball = gpGetBallForTrail()
    if not ball then
        gpTrailState.velocity = Vector3.zero
        gpTrailState.lastBall = nil
        gpTrailState.lastPos = nil
        gpTrailState.velSamples = {}
        return nil
    end
    local now = os.clock()
    if gpTrailState.lastBall ~= ball then
        gpTrailState.velSamples = {}
        gpTrailState.lastPos = ball.Position
        gpTrailState.lastTime = now
        gpTrailState.lastBall = ball
        gpTrailState.velocity = Vector3.zero
        return ball
    end
    if gpTrailState.lastPos then
        local dt = now - gpTrailState.lastTime
        if dt > 0.005 and dt < 0.3 then
            local v = (ball.Position - gpTrailState.lastPos) / dt
            if gpTrailState.velocity.Magnitude > 0 and (v - gpTrailState.velocity).Magnitude > 60 then
                gpTrailState.velSamples = {}
            else
                table.insert(gpTrailState.velSamples, v)
                if #gpTrailState.velSamples > 6 then
                    table.remove(gpTrailState.velSamples, 1)
                end
                local sum = Vector3.zero
                for _, s in ipairs(gpTrailState.velSamples) do sum = sum + s end
                if #gpTrailState.velSamples > 0 then
                    local avg = sum / #gpTrailState.velSamples
                    gpTrailState.velocity = gpTrailState.velocity:Lerp(avg, GP_VEL_SMOOTH)
                end
            end
        end
    end
    gpTrailState.lastBall = ball
    gpTrailState.lastPos = ball.Position
    gpTrailState.lastTime = now
    return ball
end

local Landing = {}

Landing.gravity = function()
    local pulled, value = pcall(function()
        local Game = require(_GP_RS.Configuration.Game)
        local State = require(_GP_RS.Common.State)
        return State.get(_GP_RS, State.Id.Modifier, "BallGravity", Game.Physics.Gravity)
    end)
    if pulled and type(value) == "number" and value > 0 then return value end
    return 17
end

Landing.velocity = function(part)
    local reached, Knit = pcall(require, _GP_RS.Packages.Knit)
    if reached then
        local found, controller = pcall(Knit.GetController, "BallController")
        if found and type(controller) == "table" and type(controller.ActiveBalls) == "table" then
            for _, record in pairs(controller.ActiveBalls) do
                local model = record.Ball
                if model and model.Parent and model:IsAncestorOf(part) and typeof(record.Velocity) == "Vector3" then
                    return record.Velocity
                end
            end
        end
    end
    return gpTrailState.velocity
end

Landing.solve = function(pos, vel, gravity, floorY)
    local drop = pos.Y - floorY
    if drop < 0 then return nil end
    local disc = vel.Y * vel.Y + 2 * gravity * drop
    if disc < 0 then return nil end
    local flight = (vel.Y + math.sqrt(disc)) / gravity
    if flight ~= flight or flight <= 0 or flight > 15 then return nil end
    return flight
end

Landing.netHit = function(pos, vel, gravity, flight)
    local map = _GP_WS:FindFirstChild("Map")
    local collide = map and map:FindFirstChild("BallCollideOnly")
    local net = collide and collide:FindFirstChild("Net")
    if not net then return false end
    local netZ = net.Position.Z
    local gap = netZ - pos.Z
    if math.abs(vel.Z) < 0.01 then return false end
    local moment = gap / vel.Z
    if moment <= 0 or moment >= flight then return false end
    local height = pos.Y + vel.Y * moment - 0.5 * gravity * moment * moment
    local across = pos.X + vel.X * moment
    local top = net.Position.Y + net.Size.Y * 0.5
    local base = net.Position.Y - net.Size.Y * 0.5
    if height > top or height < base then return false end
    return math.abs(across - net.Position.X) <= net.Size.X * 0.5
end

Landing.predict = function(ball)
    local reached, module = pcall(require, _GP_RS.Common.Physics)
    if not reached then return nil end
    local gravity = Landing.gravity()
    local pos = ball.Position
    local vel = Landing.velocity(ball)
    if typeof(vel) ~= "Vector3" then return nil end
    local floorY = module.LastFloorHeight
    if not floorY then
        local found, height = pcall(module.calculateFloorHeight, pos, true)
        floorY = found and height or nil
    end
    if not floorY then return nil end
    local flight = Landing.solve(pos, vel, gravity, floorY)
    if not flight then return nil end
    local spot = Vector3.new(pos.X + vel.X * flight, floorY, pos.Z + vel.Z * flight)
    local probed, ground = pcall(module.calculateFloorHeight, Vector3.new(spot.X, pos.Y, spot.Z), true)
    if probed and type(ground) == "number" and math.abs(ground - floorY) > 0.05 then
        local refined = Landing.solve(pos, vel, gravity, ground)
        if refined then
            flight = refined
            spot = Vector3.new(pos.X + vel.X * flight, ground, pos.Z + vel.Z * flight)
        end
    end
    return spot, module, flight, Landing.netHit(pos, vel, gravity, flight)
end

local function gpHideTrail()
    for _, p in ipairs(gpTrailState.trailParts) do
        if p and p.Parent then p.Transparency = 1 end
    end
    if gpTrailState.markerPart then gpTrailState.markerPart.Transparency = 1 end
    if gpTrailState.markerBillboard then gpTrailState.markerBillboard.Enabled = false end
end

_GP_RUN.RenderStepped:Connect(function(dt)
    if not (LHUI.BallTrajectory or LHUI.BallMarker) then
        gpHideTrail()
        return
    end
    gpEnsureTrail()
    gpEnsureMarker()
    local ball = gpUpdateBallVel()
    if not ball then
        for _, p in ipairs(gpTrailState.trailParts) do p.Transparency = 1 end
        if gpTrailState.markerPart then gpTrailState.markerPart.Transparency = 1 end
        if gpTrailState.markerBillboard then gpTrailState.markerBillboard.Enabled = false end
        return
    end
    gpTrailState.horizon = LHUI.BallHorizon or 4

    local fall, fallT = nil, nil
    if ball and gpTrailState.lastPos then
        local spot, physics, flight, netted = Landing.predict(ball)
        if spot then
            fall = spot
            fallT = flight
        end
    end
    local now = os.clock()

    if fall then
        gpTrailState.cachedFall = fall
        gpTrailState.cachedFallT = fallT
        gpTrailState.cachedFallAt = now
    end

    local useFall = fall
    local useFallT = fallT
    if not useFall and gpTrailState.cachedFall and (now - gpTrailState.cachedFallAt) < GP_CACHE_TIMEOUT then
        useFall = gpTrailState.cachedFall
        useFallT = gpTrailState.cachedFallT
    end

    if LHUI.BallTrajectory then
        local pos = gpTrailState.lastPos
        local vel = gpTrailState.velocity
        if vel.Magnitude > GP_MIN_SPEED then
            for i, p in ipairs(gpTrailState.trailParts) do
                local t = gpTrailState.horizon * i / #gpTrailState.trailParts
                p.Position = gpProjectile(pos, vel, t)
                p.Transparency = 0.45
                p.Color = LHUI.ThemeColor or Color3.fromRGB(193, 0, 255)
                p.Size = Vector3.new(0.45, 0.45, 0.45)
            end
        else
            for _, p in ipairs(gpTrailState.trailParts) do p.Transparency = 1 end
        end
    else
        for _, p in ipairs(gpTrailState.trailParts) do p.Transparency = 1 end
    end

    if LHUI.BallMarker and useFall then
        local marker = gpTrailState.markerPart
        local floorY = gpFloorY(useFall)
        local target = Vector3.new(useFall.X, floorY + 0.12, useFall.Z)

        if not gpTrailState.markerTargetPos then
            gpTrailState.markerTargetPos = target
        else
            local lerpFactor = math.clamp((dt or 0.016) * 10, 0.15, 1)
            gpTrailState.markerTargetPos = gpTrailState.markerTargetPos:Lerp(target, lerpFactor)
        end

        marker.CFrame = CFrame.new(gpTrailState.markerTargetPos) * CFrame.Angles(0, 0, math.rad(90))
        marker.Transparency = 0.35
        marker.Color = LHUI.ThemeColor or Color3.fromRGB(193, 0, 255)

        if gpTrailState.markerBillboard then
            gpTrailState.markerBillboard.Enabled = LHUI.BallMarkerTime and true or false
            if gpTrailState.markerLabel and LHUI.BallMarkerTime then
                gpTrailState.markerLabel.Text = string.format("%.2fs", useFallT or 0)
            end
        end
    elseif LHUI.BallMarker and not useFall then
        if gpTrailState.markerPart and gpTrailState.markerTargetPos then
            gpTrailState.markerPart.Transparency = 0.7
        end
    else
        if gpTrailState.markerPart then gpTrailState.markerPart.Transparency = 1 end
        if gpTrailState.markerBillboard then gpTrailState.markerBillboard.Enabled = false end
    end
end)

Protect.updateNametag = function()
    local char = Player.Character
    if not char then return end
    local nameTag = char:FindFirstChild("Nametag")
    if not nameTag then return end
    local label = nameTag:FindFirstChild("PlayerName")
    if not label or not label:IsA("TextLabel") then return end
    if Protect.enabled and Protect.customName ~= "" then
        label.Text = Protect.customName
    else
        label.Text = Player.Name
    end
end

Protect.updateTitle = function()
    if Protect.enabled and Protect.customName ~= "" then
        pcall(function()
            Player:SetAttribute("User_Title", "<font color='#FF0000'><b>Pro</b></font>")
        end)
    else
        Player:SetAttribute("User_Title", Protect.originalTitle or "")
    end
end

Protect.processNotification = function(notification)
    if not notification or not Protect.enabled or Protect.customName == "" then return end
    task.wait(0.2)
    local textLabel = notification:FindFirstChild("TextLabel")
    if not textLabel then
        for _, child in ipairs(notification:GetDescendants()) do
            if child:IsA("TextLabel") then textLabel = child break end
        end
    end
    if not textLabel then return end
    local text = textLabel.Text or ""
    if string.find(text, Player.Name, 1, true) then
        textLabel.Text = string.gsub(text, Player.Name, Protect.customName)
    end
    if string.find(text, Player.DisplayName, 1, true) then
        textLabel.Text = string.gsub(textLabel.Text, Player.DisplayName, Protect.customName)
    end
end

Protect.protectPlayerCard = function()
    if not Protect.enabled or Protect.customName == "" then return end
    local playerName = string.upper(Player.Name)
    local displayUpper = string.upper(Player.DisplayName)
    local flashFrame = Player.PlayerGui:FindFirstChild("Interface")
    if flashFrame then
        flashFrame = flashFrame:FindFirstChild("Persistent")
        if flashFrame then flashFrame = flashFrame:FindFirstChild("FlashPlayerCardFrame") end
    end
    if not flashFrame then return end
    -- Procura QUALQUER TextLabel com o nome dentro
    for _, lbl in ipairs(flashFrame:GetDescendants()) do
        if lbl:IsA("TextLabel") or lbl:IsA("TextButton") then
            local txt = lbl.Text or ""
            local up = string.upper(txt)
            if string.find(up, playerName, 1, true) then
                lbl.Text = string.gsub(txt, Player.Name, Protect.customName)
            end
            if string.find(string.upper(lbl.Text), displayUpper, 1, true) then
                lbl.Text = string.gsub(lbl.Text, Player.DisplayName, Protect.customName)
            end
        end
    end
end

    if Protect.notifConn then Protect.notifConn:Disconnect() Protect.notifConn = nil end
    if Protect.notifConn2 then Protect.notifConn2:Disconnect() Protect.notifConn2 = nil end

    Protect.notifConn = gui.ChildAdded:Connect(function(child)

        end
    end)

local AntiKick = { enabled = false, oldNamecall = nil, oldIndex = nil, installed = false }

local antiKickKeywords = {
    "kick", "ban", "anticheat", "anti_cheat", "detect", "hack", "exploit",
    "flag", "punish", "security", "stun", "ragdoll", "verify", "validate",
    "report", "sanity", "integrity", "watchdog", "monitor", "alert",
    "violation", "suspect", "speedcheck", "positioncheck", "velocitycheck",
    "flyhack", "nocliphack", "godmode", "teleport_block", "unauthorized"
}

local function isBlocked(name)
    local n = name:lower():gsub("[%s%-_]", "")
    for _, kw in ipairs(antiKickKeywords) do
        if n:find(kw, 1, true) then return true end
    end
    return false
end

local function installAntiKick()
    if AntiKick.installed then return end
    if not (getrawmetatable and setreadonly and getnamecallmethod and newcclosure) then
        print("[LEPO HUB] Anti-Kick: executor limitado")
        return
    end
    local mt = getrawmetatable(game)
    AntiKick.oldNamecall = mt.__namecall
    AntiKick.oldIndex = mt.__index
    setreadonly(mt, false)
    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if AntiKick.enabled and method == "Kick" then
            if self == Player or (Player.Character and self == Player.Character) then
                print("[LEPO HUB] Anti-Kick: Kick bloqueado!")
                return nil
            end
        end
        if AntiKick.enabled and (method == "FireServer" or method == "InvokeServer") then
            local name = tostring(self)
            if isBlocked(name) then
                print("[LEPO HUB] Anti-Kick: Remote bloqueado: " .. name:sub(1, 40))
                return nil
            end
        end
        if AntiKick.oldNamecall then return AntiKick.oldNamecall(self, ...) end
        return self[method](self, ...)
    end)
    mt.__index = newcclosure(function(self, key)
        if AntiKick.enabled and (self == Player or (Player.Character and self == Player.Character)) then
            if key == "Kick" or key == "Remove" or key == "Destroy" then
                return function() end
            end
        end
        if AntiKick.oldIndex then return AntiKick.oldIndex(self, key) end
        return nil
    end)
    setreadonly(mt, true)
    AntiKick.installed = true
end

G.jerseyEnabled = false
G.jerseySelected = "DragonTuxedoJersey"
G.jerseyOriginalClothes = {}
G.jerseyList = {}

local function gpLoadJerseyList()
    G.jerseyList = {}
    local seen = {}
    
    -- Tenta pegar do Content.Item.Entities
    local content = ReplicatedStorage:FindFirstChild("Content")
    local itemFolder = content and content:FindFirstChild("Item")
    local entities = itemFolder and itemFolder:FindFirstChild("Entities")
    
    if entities then
        for _, module in ipairs(entities:GetChildren()) do
            if module:IsA("ModuleScript") then
                local okItem, itemData = pcall(require, module)
                local itemName = module.Name
                local isJersey = string.match(itemName, "Jersey") ~= nil
                    or (okItem and type(itemData) == "table" and itemData.Type == "Jersey")
                if isJersey and not seen[itemName] then
                    seen[itemName] = true
                    table.insert(G.jerseyList, itemName)
                end
            end
        end
    end
    
    -- Fallback: pasta Assets.Jersey
    if #G.jerseyList == 0 then
        local assets = ReplicatedStorage:FindFirstChild("Assets")
        local jerseyFolder = assets and assets:FindFirstChild("Jersey")
        if jerseyFolder then
            for _, child in ipairs(jerseyFolder:GetChildren()) do
                if not seen[child.Name] then
                    seen[child.Name] = true
                    table.insert(G.jerseyList, child.Name)
                end
            end
        end
    end
    
    table.sort(G.jerseyList)
    if #G.jerseyList == 0 then
        table.insert(G.jerseyList, "DragonTuxedoJersey")
    end
    return G.jerseyList
end
gpLoadJerseyList()

-- Salva a roupa original do personagem
local function gpSaveOriginalClothes(char)
    if not char then return end
    local saved = {}
    for _, child in ipairs(char:GetChildren()) do
        if child.Name == "JerseyFront" or child.Name == "JerseyBack"
            or child:IsA("Clothing") or child:IsA("ShirtGraphic") then
            table.insert(saved, child:Clone())
        end
    end
    G.jerseyOriginalClothes[char] = saved
end

-- Restaura a roupa original
local function gpRestoreOriginalClothes(char)
    if not char then return end
    for _, child in ipairs(char:GetChildren()) do
        if child.Name == "JerseyFront" or child.Name == "JerseyBack"
            or child:IsA("Clothing") or child:IsA("ShirtGraphic") then
            child:Destroy()
        end
    end
    local saved = G.jerseyOriginalClothes[char]
    if saved then
        for _, item in ipairs(saved) do
            local clone = item:Clone()
            if clone:IsA("SurfaceGui") then
                clone.Adornee = char:FindFirstChild("UpperTorso")
            end
            clone.Parent = char
        end
        G.jerseyOriginalClothes[char] = nil
    end
end

-- Aplica a jersey no personagem
local function gpApplyJersey(jerseyId, teamName)
    local char = Player.Character
    if not char or not char:FindFirstChild("UpperTorso") then return false end
    
    local assets = ReplicatedStorage:FindFirstChild("Assets")
    assets = assets and assets:FindFirstChild("Jersey")
    if not assets then return false end
    
    -- Salva a roupa original se ainda não salvou
    if not G.jerseyOriginalClothes[char] then
        gpSaveOriginalClothes(char)
    end
    
    local okTools, jerseyTools = pcall(require, ReplicatedStorage.Tools.Jersey)
    if not okTools or type(jerseyTools) ~= "table" or not jerseyTools.set then
        return false
    end
    
    local okSet, err = pcall(jerseyTools.set, {
        Player = Player,
        Character = char,
        Id = jerseyId,
        TeamName = teamName,
    })
    
    if not okSet then
        warn("[LEPO HUB] Jersey apply failed: " .. tostring(err))
        return false
    end
    return true
end

-- Função auxiliar que detecta o time da jersey
local function gpDetectJerseyTeam()
    local char = Player.Character
    if not char then return nil end
    local shirt = char:FindFirstChildOfClass("Shirt")
    if not shirt then return nil end
    
    local assets = ReplicatedStorage:FindFirstChild("Assets")
    assets = assets and assets:FindFirstChild("Jersey")
    if not assets then return nil end
    
    for _, folder in ipairs(assets:GetChildren()) do
        for _, item in ipairs(folder:GetChildren()) do
            local shirtItem = item:FindFirstChild("Shirt")
            if shirtItem and shirtItem:IsA("Shirt") and shirtItem.ShirtTemplate == shirt.ShirtTemplate then
                return item.Name
            end
        end
    end
    return nil
end
